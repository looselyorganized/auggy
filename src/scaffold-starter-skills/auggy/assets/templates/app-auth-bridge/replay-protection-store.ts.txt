import type { ExternalAuthReplayStore } from "auggy";

type SetOnceWithTtlMs = (key: string, value: string, ttlMs: number) => Promise<boolean>;

export function createSharedExternalAuthReplayStore(
  setOnceWithTtlMs: SetOnceWithTtlMs,
): ExternalAuthReplayStore {
  return {
    async consume(jti, expiresAt, now) {
      const ttlMs = Math.max(0, expiresAt - now);
      if (ttlMs <= 0) return false;
      return setOnceWithTtlMs(`auggy:external-auth:jti:${jti}`, "1", ttlMs);
    },
  };
}
