import { auth, currentUser } from "@clerk/nextjs/server";
import { createExternalAuthAssertion } from "auggy";
import { deriveAuggyAuthorization, type VerifiedAppSession } from "./app-policy";
import {
  assertionJson,
  assertionMethodNotAllowed,
  requireCookieAssertionRequest,
} from "./assertion-response";

export function GET() {
  return assertionMethodNotAllowed();
}

export async function POST(request: Request) {
  const boundaryError = requireCookieAssertionRequest(request);
  if (boundaryError) return boundaryError;

  try {
    const session = await verifyClerkSession();
    if (!session) return assertionJson({ error: "unauthorized" }, 401);

    const authorization = deriveAuggyAuthorization(session);
    const assertion = createExternalAuthAssertion({
      secret: process.env.AUGGY_EXTERNAL_AUTH_SECRET!,
      keyId: process.env.AUGGY_EXTERNAL_AUTH_KEY_ID ?? "2026-07",
      audience: process.env.AUGGY_EXTERNAL_AUTH_AUDIENCE!,
      provider: "clerk",
      subject: session.userId,
      ttlSeconds: 60,
      email: session.email,
      emailVerified: session.emailVerified,
      orgId: session.orgId,
      roles: session.roles,
      scopes: authorization.scopes,
      grants: authorization.grants,
      authzVersion: "2026-07-08",
      jti: crypto.randomUUID(),
    });

    return assertionJson({ assertion });
  } catch {
    return assertionJson({ error: "assertion_unavailable" }, 500);
  }
}

async function verifyClerkSession(): Promise<VerifiedAppSession | null> {
  const { isAuthenticated, orgId, sessionClaims, userId } = await auth();
  if (!isAuthenticated || !userId) return null;

  const user = await currentUser();

  return {
    provider: "clerk",
    userId,
    email: user?.primaryEmailAddress?.emailAddress,
    emailVerified: user?.primaryEmailAddress?.verification?.status === "verified",
    orgId: orgId ?? undefined,
    roles: rolesFromClaims(sessionClaims),
    orderIds: await loadReadableOrderIds(userId),
  };
}

function rolesFromClaims(claims: unknown): string[] {
  if (!claims || typeof claims !== "object") return [];
  const metadata = "metadata" in claims ? (claims as { metadata?: unknown }).metadata : undefined;
  if (!metadata || typeof metadata !== "object") return [];
  const roles = "roles" in metadata ? (metadata as { roles?: unknown }).roles : undefined;
  return Array.isArray(roles) ? roles.filter((item): item is string => typeof item === "string") : [];
}

async function loadReadableOrderIds(_userId: string): Promise<string[]> {
  return [];
}
